configurar o repositorio local apenas uma vez e sempre executar os comandos na pasta do projeto
----------
//iniciar projeto localmente em pasta vazia
git init

//configuração do perfil
git config --global user.name "SeuNome"
git config --global user.email "seuemail@exemplo.com"

//conectar repositório
git remote add origin https://github.com/seuusuario/nome-do-repo.git

----------usa uma pasta só pra branch que tu vai mexer

//cria uma branch nova e muda automaticamente para ela
git checkout -b <nome>

//trocar localização da branch
git checkout <nome>

//verifica qual branch tu ta
git status

//setupstream para enviar arquivos pela primeira vez
git push --set-upstream origin <nome>

//atualizar arquivos
git add . && git commit -m "<nomeAtt>" && git push

//faz atualização à força
git push origin <nome_da_branch> --force

//mandar o git ignorar arquivos
git rm --cached -r <nome_da_pasta>/
